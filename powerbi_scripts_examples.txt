import sys
sys.path.insert(0, r'C:\Users\LENOVO\Desktop\assistant ai\src')

import pandas as pd
from outils_donnees import comparer_segments

# Comparer France vs Germany
result = comparer_segments('France', 'Germany')

if result.get('donnees_trouvees'):
    # Créer un DataFrame avec les deux segments côte à côte
    dataset = pd.DataFrame([
        {
            'segment': 'France',
            'nombre_clients': result['segment_a']['nombre_clients'],
            'rentabilite_moyenne': result['segment_a']['rentabilite_moyenne'],
            'pnb_moyen': result['segment_a']['pnb_moyen'],
            'cout_moyen': result['segment_a']['cout_moyen']
        },
        {
            'segment': 'Germany',
            'nombre_clients': result['segment_b']['nombre_clients'],
            'rentabilite_moyenne': result['segment_b']['rentabilite_moyenne'],
            'pnb_moyen': result['segment_b']['pnb_moyen'],
            'cout_moyen': result['segment_b']['cout_moyen']
        }
    ])
    
    # Ajouter les différences
    dataset['diff_rentabilite'] = [
        result['differences']['rentabilite_moyenne'], 
        -result['differences']['rentabilite_moyenne']
    ]
else:
    dataset = pd.DataFrame({'erreur': ['Impossible de comparer']})

SCRIPT 10: ÉVOLUTION PAR PÉRIODE (Si périodes disponibles)
=================================================================

import sys
sys.path.insert(0, r'C:\Users\LENOVO\Desktop\assistant ai\src')

import pandas as pd
import sqlite3

db_path = r'C:\Users\LENOVO\Desktop\assistant ai\data\cdg_test.db'
conn = sqlite3.connect(db_path)

# Agrégation par segment et période
query = """
    SELECT 
        segment,
        periode,
        COUNT(*) as nombre_clients,
        ROUND(SUM(pnb_simule), 2) as pnb_total,
        ROUND(AVG(rentabilite), 2) as rentabilite_moyenne
    FROM pnb_analytique
    GROUP BY segment, periode
    ORDER BY segment, periode
"""

dataset = pd.read_sql_query(query, conn)
conn.close()

# Colonnes: segment, periode, nombre_clients, pnb_total, 
#          rentabilite_moyenne
